#include <iostream>
using namespace std;

int main()
{
    int checkpoints;
    double position;
    double time;
    double previousPosition = 0;
    double previousTime = 0;
    double firstPosition = 0;
    double firstTime = 0;
    double distance;
    double timePassed;
    double speed;

    cout << "How many checkpoints would you like to record? ";
    cin >> checkpoints;

    while (checkpoints < 2)
    {
        cout << "Error: You must enter at least 2 checkpoints." << endl;
        cout << "How many checkpoints would you like to record? ";
        cin >> checkpoints;
    }

    for (int count = 1; count <= checkpoints; count++)
    {
        cout << "Enter position (miles) and time (hours) for checkpoint "
            << count << ": ";
        cin >> position >> time;

        if (count == 1)
        {
            firstPosition = position;
            firstTime = time;
        }

        while (count > 1 && time <= previousTime)
        {
            cout << "Error: Time must be greater than the previous checkpoint." << endl;
            cout << "Enter position (miles) and time (hours) for checkpoint "
                << count << ": ";
            cin >> position >> time;
        }

        if (count > 1)
        {
            distance = position - previousPosition;
            timePassed = time - previousTime;
            speed = distance / timePassed;

            cout << "Average speed, checkpoint " << count - 1
                << " to " << count << ": " << speed << " mph" << endl;
        }

        previousPosition = position;
        previousTime = time;
    }

    distance = previousPosition - firstPosition;
    timePassed = previousTime - firstTime;
    speed = distance / timePassed;

    cout << "Overall average speed for the trip: "
        << speed << " mph" << endl;

    return 0;
}
