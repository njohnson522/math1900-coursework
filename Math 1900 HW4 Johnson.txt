#include <iostream>

using namespace std;
//finding midpoints of coordinates 

int main()
{
    string name;
    double x1, y1, x2, y2;
    double midpoint_x, midpoint_y;

    cout << "Enter your name: ";
    getline(cin, name);

    cout << "Enter x1: ";
    cin >> x1;

    cout << "Enter y1: ";
    cin >> y1;

    cout << "Enter x2: ";
    cin >> x2;

    cout << "Enter y2: ";
    cin >> y2;

    midpoint_x = (x1 + x2) / 2;
    midpoint_y = (y1 + y2) / 2;

    cout << "\n" << name << ", the midpoint is: ("
        << midpoint_x << ", " << midpoint_y << ")" << endl;

    return 0;
}
