// Math 1900 HW10 Johnson
//1.Create constants for the number of rows(3) and columns(4).
//2.Declare a 2D array with the 12 given numbers arranged in 3 rows and 4 columns.
//3.Use nested for loops to go through the array and print the numbers in a table.Use setw() to line up the columns.
//4.Use another set of nested for loops to calculate the total of each row.Reset the total to 0 for each new row.
//5.Print each row's total.
//6.Use another set of nested for loops to calculate the total of each column.Reset the total to 0 for each new column.
//7.Print each column's total.
#include <iostream>
#include <iomanip>
using namespace std;

int main() {

    const int ROWS = 3;
    const int COLS = 4;

    int numbers[ROWS][COLS] = {
        {1, 2, 3, 4},
        {2, 4, 6, 8},
        {3, 6, 9, 12}
    };

    cout << "Table:" << endl;

    for (int row = 0; row < ROWS; row++) {
        for (int col = 0; col < COLS; col++) {
            cout << setw(5) << numbers[row][col];
        }
        cout << endl;
    }

    cout << "\nRow Totals:" << endl;

    for (int row = 0; row < ROWS; row++) {
        int total = 0;

        for (int col = 0; col < COLS; col++) {
            total += numbers[row][col];
        }

        cout << "Row " << row + 1 << ": " << total << endl;
    }

    cout << "\nColumn Totals:" << endl;

    for (int col = 0; col < COLS; col++) {
        int total = 0;

        for (int row = 0; row < ROWS; row++) {
            total += numbers[row][col];
        }

        cout << "Column " << col + 1 << ": " << total << endl;
    }

    return 0;
}

// Math 1900 HW10 Johnson
//1. Create constants above main() for the number of students(2) and quizzes(3).
//2. Write the function prototype for rowAverage above main().
//3. Create a 2D array of doubles with 2 rows and 3 columns to store the quiz scores.
//4. Use nested for loops to ask the user to enter all six quiz scores into the array.
//5. Use a for loop to go through each student and call the rowAverage function to calculate and print their average.
//6. Use nested for loops to calculate the average score for each quiz(column).
//7. Print the average for each quiz.
//8. Write the rowAverage function below main().Have it add the scores in the selected row, divide by the number of quizzes, and return the average.
#include <iostream>
#include <iomanip>
using namespace std;

const int STUDENTS = 2;
const int QUIZZES = 3;

double rowAverage(double scores[STUDENTS][QUIZZES], int row);

int main() {

    double scores[STUDENTS][QUIZZES];

    for (int row = 0; row < STUDENTS; row++) {
        for (int col = 0; col < QUIZZES; col++) {
            cout << "Enter score for Student " << row + 1;
            cout << ", Quiz " << col + 1 << ": ";
            cin >> scores[row][col];
        }
    }

    cout << fixed << setprecision(2);

    cout << "\nStudent Averages:" << endl;

    for (int row = 0; row < STUDENTS; row++) {
        cout << "Student " << row + 1 << ": ";
        cout << rowAverage(scores, row) << endl;
    }

    cout << "\nQuiz Averages:" << endl;

    for (int col = 0; col < QUIZZES; col++) {
        double total = 0;

        for (int row = 0; row < STUDENTS; row++) {
            total += scores[row][col];
        }

        cout << "Quiz " << col + 1 << ": ";
        cout << total / STUDENTS << endl;
    }

    return 0;
}

double rowAverage(double scores[STUDENTS][QUIZZES], int row) {

    double total = 0;

    for (int col = 0; col < QUIZZES; col++) {
        total += scores[row][col];
    }

    return total / QUIZZES;
}
