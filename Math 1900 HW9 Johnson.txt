// Use a for loop to display the values with a space in between, all on the same line.
//Use a second for loop to display the numbers, each on its own line.
//Use a third for loop to display the numbers in reverse order, all on the same line
// 1. Declare int size and numbers with my arrays values
// 2. create a for loop that displays each number on the same line
// 3. create a for loop that displays each number on its own line
// 4. create a for loop that lists numbers in array backwards
#include <iostream>
using namespace std;

int main() {

    const int SIZE = 5;
    int numbers[SIZE] = { 2, 4, 16, 256, 65536 };

    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;

    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << endl;
    }

    for (int i = SIZE - 1; i >= 0; i--) {
        cout << numbers[i] << " ";
    }
    cout << endl;

    return 0;
}

// Math 1900 HW9 Johnson
// Plan,Goal
//create a program that finds order total with 2  arrays
//1. create a function that bundles a returns prices, quanities and size
//2. create arrays for prices and quantities and the values at each position
//3. create for loop that displays the product of  quantites and price for each 
//4. declare double for total and display total on new line
//5. create for loop to find total
#include <iostream>
using namespace std;

double orderTotal(double prices[], int quantities[], int size);

int main() {

    const int SIZE = 3;

    double prices[SIZE] = { 5.50, 8.25, 11.85 };
    int quantities[SIZE] = { 2, 4, 8 };

    for (int i = 0; i < SIZE; i++) {
        cout << "Quantity: " << quantities[i] << "  x  "
            << " Price: $" << prices[i]
            << " = $" << prices[i] * quantities[i] << endl;
    }

    double total = orderTotal(prices, quantities, SIZE);

    cout << "Order total: $" << total << endl;

    return 0;
}

double orderTotal(double prices[], int quantities[], int size) {

    double total = 0;

    for (int i = 0; i < size; i++) {
        total = total + (prices[i] * quantities[i]);
    }

    return total;
}
