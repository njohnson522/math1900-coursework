// Define an array with the following 5 values: {0, 10, 2, 4, 10}. Then:
// Use a for loop to display the values with a space in between, all on the same line.
//Use a second for loop to display the numbers, each on its own line.
//Use a third for loop to display the numbers in reverse order, all on the same line
// 1. Declare int size and numbers with my arrays values
// 2. create a for loop that displays each number on the same line
// 3. create a for loop that displays each number on its own line
// 4. create a for loop that lists numbers in array backwards
#include <iostream>
using namespace std;

int main() {

    const int SIZE = 5;
    int numbers[SIZE] = { 2, 4, 16, 256, 65536 };

    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;

    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << endl;
    }

    for (int i = SIZE - 1; i >= 0; i--) {
        cout << numbers[i] << " ";
    }
    cout << endl;

    return 0;
}
Problem 2
// Math 1900 HW9 Johnson
// Plan,Goal
// Create a program that allows a user to enter 3 values and them display them back
//1. declare int size and double numbers with the array
//2. create a for loop that gathers the 3 doubles from user for array
//3. create for loop that displays each number of array built by user
#include <iostream>
using namespace std;

int main() {

    const int SIZE = 3;
    double numbers[SIZE];

    for (int i = 0; i < SIZE; i++) {
        cout << "Enter a number: ";
        cin >> numbers[i];
    }

    for (int i = 0; i < SIZE; i++) {
        cout << numbers[i] << " ";
    }

    cout << endl;

    return 0;
}
